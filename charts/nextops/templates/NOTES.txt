  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "nextops.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")

{{- else if contains "LoadBalancer" .Values.service.type }}

{{- else }}
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "nextops.fullname" . }} 3000:{{ .Values.service.port }}
  echo "Visit http://127.0.0.1:3000"
{{- end }}

2. Before production use, verify secrets were overridden (not CHANGEME placeholders).
   Direct Upstream Access (ADR-0019): PUBLIC_DEFAULT_PROXY_ENABLED and
   PUBLIC_MYSQL_PROXY_ENABLED default to false. Opt a switch in only via values.

3. Log in and open SRE admin at /admin (redirects to /admin/sources). Dashboard refresh and pipeline thresholds are per-browser settings under /admin/dashboard.

4. For SSE endpoints (/api/alerts/stream, /api/logs/stream), disable proxy buffering on your Ingress controller.
